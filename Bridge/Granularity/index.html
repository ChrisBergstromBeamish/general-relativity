c = s.Matrix([s.cos(theta/2), s.exp(s.I*phi)*s.sin(theta/2)])
dc = [c.diff(theta), c.diff(phi)]
inner = lambda u, v: (s.conjugate(u).T*v)[0]
clean = lambda x: s.simplify(s.expand_complex(x))
checks = {}
def zero(name, expr):
    residual = s.simplify(s.trigsimp(s.expand_trig(expr)))
    if residual != 0:
        # Clear chart denominators before simplifying trigonometric numerators.
        # An identity on the regular chart extends to these smooth sphere modes.
        numerator = s.together(expr).as_numer_denom()[0]
        residual = s.simplify(s.trigsimp(s.expand(numerator)))
    if residual != 0:
        raise AssertionError((name, residual))
    checks[name] = 'exact residual 0'

zero('normalized coefficient field', clean(inner(c,c))-1)
a = [clean(-s.I*inner(c, v)) for v in dc]
zero('north potential theta component', a[0])
zero('north potential phi component', a[1]-(1-s.cos(theta))/2)
curv = s.diff(a[1],theta)-s.diff(a[0],phi)
zero('curvature from phase comparison', curv-s.sin(theta)/2)
G = clean(2*Q*s.im(inner(dc[0],dc[1])))
zero('tensor from coefficient variations', G-Q*s.sin(theta)/2)
aS = -(1+s.cos(theta))/2
zero('north south gluing', a[1]-aS-1)
zero('south chart same tensor', s.diff(aS,theta)-curv)
c3=s.Matrix([s.sqrt(s.binomial(3,r))*s.cos(theta/2)**(3-r)*s.sin(theta/2)**r*s.exp(s.I*r*phi) for r in range(4)])
zero('spin three halves coherent orbit norm',clean(inner(c3,c3))-1)
zero('spin three halves orbit phase factor',clean(-s.I*inner(c3,c3.diff(phi)))-3*a[1])
flux = s.integrate(s.integrate(Q*curv,(phi,0,2*s.pi)),(theta,0,s.pi))
zero('whole orientation sphere period', flux-2*s.pi*Q)
cap = s.integrate(s.integrate(Q*curv,(phi,0,2*s.pi)),(theta,0,s.pi/3))
zero('60 degree cap action', cap-s.pi*Q/2)

# Horizontal overlap metric is used only to check continuity, not as a grain floor.
metric = s.Matrix(2,2,lambda i,j:clean(s.re(inner(dc[i],dc[j])-inner(dc[i],c)*inner(c,dc[j]))))
zero('orientation metric theta',metric[0,0]-s.Rational(1,4))
zero('orientation metric phi',metric[1,1]-s.sin(theta)**2/4)
zero('orientation metric mixed',metric[0,1])

S = s.Matrix([Q*s.sin(theta)*s.cos(phi)/2,Q*s.sin(theta)*s.sin(phi)/2,Q*s.cos(theta)/2])
def bracket(u,v):
    return s.trigsimp(2*(s.diff(u,theta)*s.diff(v,phi)-s.diff(u,phi)*s.diff(v,theta))/(Q*s.sin(theta)))
for i,j,k in [(0,1,2),(1,2,0),(2,0,1)]:

    return (s.diff(s.sin(eta)*s.cos(eta)*s.diff(fun,eta),eta)/(s.sin(eta)*s.cos(eta))
            +s.diff(fun,alpha,2)/s.cos(eta)**2+s.diff(fun,beta,2)/s.sin(eta)**2)
low = s.sqrt(2)*s.cos(eta)*s.exp(-s.I*alpha)
